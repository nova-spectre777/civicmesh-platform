cmake_minimum_required(VERSION 3.20)
project(civicmesh_edge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(civicmesh_edge src/mesh.cpp)
target_include_directories(civicmesh_edge PUBLIC include)
add_executable(civicmesh-edge src/main.cpp)
target_link_libraries(civicmesh-edge PRIVATE civicmesh_edge)
add_executable(civicmesh-edge-tests tests/test_mesh.cpp)
target_link_libraries(civicmesh-edge-tests PRIVATE civicmesh_edge)
enable_testing()
add_test(NAME edge-tests COMMAND civicmesh-edge-tests)
